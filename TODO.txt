fill $3841,1,$F0

replace division: 

bpset FED8,,{temp0=A;A /= B; B = A % B;PC=FEEF;g}
bpset FEF0,,{temp0=w@U; temp1=temp0/D; w@U=temp1; D = temp0%D; PC=$feef; g}

record sounds: bpset 791D,B != $FF,{B=$E;g}
